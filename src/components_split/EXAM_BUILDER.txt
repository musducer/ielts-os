        {activeTab === "EXAM_BUILDER" && (() => {

            const currentPath = normalizeFolderPath(builderFolder);

            allKnownFolders.forEach(rawF => {

                        childQuizzes.push(q);

                    if (qPath === currentPath && matchesSkill(q)) {
                        childQuizzes.push(q);

                }

                if (builderSort === "TITLE_ASC") {
                    return (a.title || "").localeCompare(b.title || "");

                if (builderSort === "NEWEST") {
                    return Number(b.updatedAt || b.createdAt || b.id || 0) - Number(a.updatedAt || a.createdAt || a.id || 0);

                if (builderSort === "QUESTIONS") {
                    return (b.questions?.length || 0) - (a.questions?.length || 0);

                    if (q.type === 'CHOICE_MULTIPLE' || q.type === 'MATCHING' || q.type === 'BLANK' || q.type === 'DRAG_DROP' || q.type === 'SHORT_ANSWER' || q.type === 'DRAG_DROP_HEADING' || q.type === 'MAP_DRAG' || q.type === 'DIAGRAM_LABEL') {

                        while (j < qs.length && qs[j].type === q.type

                            j++;
                        }
                        if (sharedQs.length > 1) {
                            builderGroups.push({ isMerged: true, questions: sharedQs, startIndex: i, groupType: q.type });
                            i = j - 1;
                            continue;
                        }

                    builderGroups.push({ isMerged: false, questions: [q], startIndex: i, groupType: q.type });

                                                <Ico name="folder" size={13} /> {t('eb_move_folder') || 'Move to…'}
                                            </button>

                                            <button className="ebx-soft" onClick={() => typeof createRealExamPackageFromSelection === 'function' && createRealExamPackageFromSelection()} style={{background: 'transparent', color: EB.ink, padding: '5px 9px', fontSize: 12, border: `1px solid ${EB.line}`, borderRadius: 6, fontWeight: 600, cursor: 'pointer'}}>
                                                Real package
                                            </button>
                                        </div>

                                        <button className="ebx-soft" onClick={() => typeof handleBulkDeleteQuizzes === 'function' && handleBulkDeleteQuizzes()} style={{background: `${C.err}0a`, color: C.err, padding: '5px 10px', fontSize: 12, border: `1px solid ${C.err}33`, borderRadius: 6, fontWeight: 600, cursor: 'pointer', display: 'inline-flex', alignItems: 'center', gap: 5}}>
                                            <Ico name="trash" size={13} /> {t('eb_delete') || 'Delete'}
                                        </button>
                                    </div>
                                )}

                                <div style={{display: 'grid', gap: 8}}>
                                    {childQuizzes.map((q: any) => {
                                        const isSelected = selectedQuizzes.includes(q.id);
                                        const ghostBtn: any = {background: EB.sheet, color: EB.sub, padding: '6px 10px', fontSize: 12, border: `1px solid ${EB.line}`, borderRadius: EB.radiusSm, fontWeight: 600, cursor: 'pointer', display: 'inline-flex', alignItems: 'center', gap: 5};
                                        return (
                                            <div
                                                key={q.id}
                                                className="ebx-card"
                                                draggable
                                                onDragStart={(e) => {
                                                    e.dataTransfer.setData("application/json", JSON.stringify({ type: "QUIZ", quizId: q.id }));
                                                    e.dataTransfer.effectAllowed = "move";
                                                }}
                                                style={{
                                                    background: isSelected ? EB.accentWash : EB.sheet,
                                                    padding: '12px 16px',
                                                    borderRadius: EB.radiusSm,
                                                    border: `1px solid ${isSelected ? EB.accent : EB.line}`,
                                                    borderLeft: `3px solid ${isSelected ? EB.accent : EB.line}`,
                                                    display: 'flex',
                                                    justifyContent: 'space-between',
                                                    alignItems: 'center',
                                                    flexWrap: 'wrap',
                                                    gap: 14,
                                                    transition: 'border-color 0.15s, background-color 0.15s'
                                                }}
                                            >
                                                <div style={{display: 'flex', gap: 12, alignItems: 'center', minWidth: 0, flex: '1 1 360px'}}>
                                                    <input
                                                        type="checkbox"
                                                        checked={isSelected}
                                                        onChange={(e: any) => {
                                                            if(e.target.checked) setSelectedQuizzes([...selectedQuizzes, q.id]);
                                                            else setSelectedQuizzes(selectedQuizzes.filter((id: any) => id !== q.id));
                                                        }}
                                                        style={{width: 16, height: 16, cursor: 'pointer', accentColor: EB.accent, flexShrink: 0}}
                                                    />
                                                    <div style={{width: 36, height: 36, borderRadius: 8, background: EB.wash, display: 'grid', placeItems: 'center', color: q.type === 'Listening' ? EB.accent : EB.ink, flexShrink: 0}}>
                                                        {q.type === 'Listening' ? <Ico name="headphones" size={18} color={EB.accent} /> : <Ico name="book" size={18} color={EB.ink} />}
                                                    </div>
                                                    <div style={{minWidth: 0, flex: 1}}>
                                                        <div style={{display: 'flex', alignItems: 'center', flexWrap: 'wrap', gap: 6, marginBottom: 2}}>
                                                            <span style={{fontFamily: EB.fDisplay, fontWeight: 600, fontSize: 15, color: EB.ink, letterSpacing: '-0.01em', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap'}}>
                                                                {q.title}
                                                            </span>

                                                            <span style={{display: 'inline-flex', verticalAlign: 'middle', padding: '1px 6px', borderRadius: 999, border: `1px solid ${getDeliveryMode(q) === 'practice' ? C.succ : EB.line}`, color: getDeliveryMode(q) === 'practice' ? C.succ : EB.sub, fontFamily: EB.fMono, fontSize: 9.5, letterSpacing: '.04em'}}>
                                                                {getDeliveryMode(q).toUpperCase()}
                                                            </span>
                                                            <span style={{display: 'inline-flex', verticalAlign: 'middle', padding: '1px 6px', borderRadius: 4, background: q.active ? `${C.succ}15` : EB.wash, color: q.active ? C.succ : EB.sub, fontFamily: EB.fLabel, fontSize: 9.5, fontWeight: 700, letterSpacing: '0.06em', textTransform: 'uppercase'}}>
                                                                {q.active ? 'Published' : 'Draft'}
                                                            </span>
                                                        </div>
                                                        <div style={{fontSize: 12, color: EB.sub, fontWeight: 500, display: 'flex', alignItems: 'center', gap: 6, flexWrap: 'wrap'}}>
                                                            <span>{q.type}</span>
                                                            <span style={{opacity: .4}}>·</span>
                                                            <span><span style={{fontFamily: EB.fMono}}>{q.timeLimit}</span> {t('eb_minutes')}</span>
                                                            <span style={{opacity: .4}}>·</span>
                                                            <span><span style={{fontFamily: EB.fMono}}>{(q.questions || []).length}</span> {t('eb_questions_short')}</span>

                                                            {builderSearch && q.folder && (
                                                                <>
                                                                    <span style={{opacity: .4}}>·</span>
                                                                    <span style={{fontFamily: EB.fMono, fontSize: 10.5, color: EB.sub}}><Ico name="folder" size={11} style={{verticalAlign: '-1px', marginRight: 3}} />{q.folder}</span>
                                                                </>
                                                            )}
                                                        </div>
                                                    </div>
                                                </div>
                                                <div style={{display: 'flex', gap: 6, flexWrap: 'wrap', alignItems: 'center'}}>
                                                    {typeof startExam === 'function' && <button className="ebx-soft" onClick={() => startExam(q, true, false)} style={ghostBtn}><Ico name="eye" size={13} /> {t('eb_preview_exam')}</button>}
                                                    <button className="ebx-soft" onClick={() => setKeyEditingQuiz(q)} style={{...ghostBtn, color: EB.warn, borderColor: `${C.warn}33`}}><Ico name="key" size={13} /> {t('eb_edit_key_btn')}</button>
                                                    {typeof duplicateQuiz === 'function' && <button className="ebx-soft" onClick={() => duplicateQuiz(q)} style={ghostBtn}><Ico name="copy" size={13} /> {t('eb_duplicate')}</button>}
                                                    {typeof handleExportExamKey === 'function' && <button className="ebx-soft" onClick={() => handleExportExamKey(q)} style={ghostBtn}><Ico name="download" size={13} /> Key</button>}
                                                    {typeof handleRecalculateScores === 'function' && <button className="ebx-soft" onClick={() => handleRecalculateScores(q.id)} title={t('eb_recalc')} style={ghostBtn}><Ico name="refresh" size={13} /></button>}
                                                    <button className="ebx-primary" onClick={() => setEditingQuiz(JSON.parse(JSON.stringify(q)))} style={{background: EB.accent, color: '#fff', padding: '6px 14px', fontSize: 12.5, border: 'none', borderRadius: EB.radiusSm, fontWeight: 700, cursor: 'pointer', display: 'inline-flex', alignItems: 'center', gap: 6}}>
                                                        <Ico name="edit" size={13} /> {t('eb_edit_exam')}
                                                    </button>
                                                    {typeof handleDeleteSingleQuiz === 'function' && (
                                                        <button
                                                            className="ebx-soft"
                                                            onClick={() => handleDeleteSingleQuiz(q.id, q.title)}
                                                            title={tx('Xóa đề thi', 'Delete exam')}
                                                            style={{...ghostBtn, color: EB.err, borderColor: `${C.err}33`}}
                                                        >
                                                            <Ico name="trash" size={13} />
                                                        </button>
                                                    )}
                                                </div>
                                            </div>
                                        );
                                    })}
                                    {childQuizzes.length === 0 && (
                                        <div style={{
                                            textAlign: 'center',
                                            color: EB.sub,
                                            padding: '48px 24px',
                                            background: EB.sheet,
                                            borderRadius: EB.radius,
                                            border: `1px dashed ${EB.line}`,
                                            display: 'flex',
                                            flexDirection: 'column',
                                            alignItems: 'center',
                                            gap: 8
                                        }}>
                                            <Ico name="file" size={26} color={EB.sub} />
                                            <div style={{fontFamily: EB.fDisplay, fontSize: 16, fontWeight: 600, color: EB.ink}}>
                                                {builderSearch ? tx(`Không tìm thấy đề thi phù hợp với "${builderSearch}"`, `No exams matching "${builderSearch}"`) : (t('eb_folder_empty') || tx('Thư mục này chưa có đề thi', 'This folder is empty'))}
                                            </div>
                                            <div style={{fontSize: 12.5, color: EB.sub, maxWidth: 360, lineHeight: 1.5}}>
                                                {builderSearch ? tx('Thử tìm với từ khóa khác hoặc xóa bộ lọc để xem tất cả đề thi.', 'Try different keywords or clear filters to see all exams.') : tx('Tạo đề mới hoặc kéo thả đề thi từ các thư mục khác vào đây.', 'Create a new exam or drag and drop exams from other folders here.')}
                                            </div>
                                        </div>
                                    )}
                                </div>
                            </div>
                        </div>
                    </div>
                )}

                {/* MODAL FILE MANAGER: TẠO / ĐỔI TÊN / CHUYỂN THƯ MỤC / CHUYỂN ĐỀ THI */}
                {folderModal && (
                    <div
                        role="dialog"
                        aria-modal="true"
                        onClick={() => setFolderModal(null)}
                        style={{
                            position: 'fixed',
                            inset: 0,
                            background: 'rgba(26, 23, 38, 0.45)',
                            backdropFilter: 'blur(4px)',
                            zIndex: 99999,
                            display: 'grid',
                            placeItems: 'center',
                            padding: 16
                        }}
                    >
                        <div
                            onClick={(e) => e.stopPropagation()}
                            style={{
                                width: '100%',
                                maxWidth: 480,
                                background: EB.sheet,
                                border: `1px solid ${EB.line}`,
                                borderRadius: EB.radius,
                                boxShadow: '0 24px 48px rgba(26, 23, 38, 0.18)',
                                overflow: 'hidden',
                                display: 'flex',
                                flexDirection: 'column'
                            }}
                        >
                            {/* Modal Header */}
                            <div style={{
                                padding: '16px 20px',
                                borderBottom: `1px solid ${EB.line}`,
                                display: 'flex',
                                alignItems: 'center',
                                justifyContent: 'space-between',
                                background: EB.wash
                            }}>
                                <div style={{display: 'flex', alignItems: 'center', gap: 8}}>
                                    <Ico name="folder" size={16} color={EB.accent} />
                                    <span style={{fontFamily: EB.fDisplay, fontWeight: 700, fontSize: 16, color: EB.ink}}>
                                        {folderModal.mode === "CREATE" && (t('eb_new_folder') || tx("Tạo thư mục mới", "Create new folder"))}
                                        {folderModal.mode === "RENAME" && tx("Đổi tên thư mục", "Rename folder")}
                                        {folderModal.mode === "MOVE_FOLDER" && tx("Di chuyển thư mục", "Move folder")}
                                        {folderModal.mode === "MOVE_QUIZZES" && tx(`Di chuyển ${selectedQuizzes.length} đề thi`, `Move ${selectedQuizzes.length} exams`)}
                                    </span>
                                </div>
                                <button
                                    type="button"
                                    onClick={() => setFolderModal(null)}
                                    aria-label={tx("Đóng", "Close")}
                                    style={{
                                        background: 'transparent',
                                        border: 'none',
                                        color: EB.sub,
                                        cursor: 'pointer',
                                        padding: 4,
                                        borderRadius: 4,
                                        display: 'inline-flex'
                                    }}
                                >
                                    <Ico name="x" size={16} />
                                </button>
                            </div>

                            {/* Modal Body */}
                            <div style={{padding: '20px'}}>
                                {folderModal.mode === "CREATE" && (
                                    <div>
                                        <label style={{display: 'block', fontFamily: EB.fLabel, fontSize: 11, fontWeight: 700, letterSpacing: '0.08em', textTransform: 'uppercase', color: EB.sub, marginBottom: 8}}>
                                            {tx("Tên thư mục", "Folder name")}
                                        </label>
                                        <input
                                            autoFocus
                                            type="text"
                                            value={folderModal.inputValue}
                                            placeholder={tx("Nhập tên thư mục (VD: Test 1, Unit 2...)", "Enter folder name (e.g. Test 1, Unit 2...)")}
                                            onChange={(e) => setFolderModal({ ...folderModal, inputValue: e.target.value })}
                                            onKeyDown={(e) => {
                                                if (e.key === "Enter" && folderModal.inputValue.trim()) {
                                                    handleCreateFolder(folderModal.inputValue, folderModal.targetFolder || builderFolder);
                                                    setFolderModal(null);
                                                }
                                            }}
                                            style={{
                                                width: '100%',
                                                padding: '10px 14px',
                                                borderRadius: EB.radiusSm,
                                                border: `1px solid ${EB.line}`,
                                                background: EB.paper,
                                                color: EB.ink,
                                                fontSize: 14,
                                                fontFamily: EB.fBody,
                                                outline: 'none',
                                                boxSizing: 'border-box'
                                            }}
                                        />
                                        <div style={{fontFamily: EB.fMono, fontSize: 11, color: EB.sub, marginTop: 8}}>
                                            {tx('Vị trí:', 'Location:')} <span style={{color: EB.ink, fontWeight: 600}}>{folderModal.targetFolder || builderFolder}</span>
                                        </div>
                                    </div>
                                )}

                                {folderModal.mode === "RENAME" && (
                                    <div>
                                        <label style={{display: 'block', fontFamily: EB.fLabel, fontSize: 11, fontWeight: 700, letterSpacing: '0.08em', textTransform: 'uppercase', color: EB.sub, marginBottom: 8}}>
                                            {tx("Tên mới", "New name")}
                                        </label>
                                        <input
                                            autoFocus
                                            type="text"
                                            value={folderModal.inputValue}
                                            placeholder={tx("Tên thư mục mới", "New folder name")}
                                            onChange={(e) => setFolderModal({ ...folderModal, inputValue: e.target.value })}
                                            onKeyDown={(e) => {
                                                if (e.key === "Enter" && folderModal.inputValue.trim() && folderModal.targetFolder) {
                                                    handleRenameFolder(folderModal.targetFolder, folderModal.inputValue);
                                                    setFolderModal(null);
                                                }
                                            }}
                                            style={{
                                                width: '100%',
                                                padding: '10px 14px',
                                                borderRadius: EB.radiusSm,
                                                border: `1px solid ${EB.line}`,
                                                background: EB.paper,
                                                color: EB.ink,
                                                fontSize: 14,
                                                fontFamily: EB.fBody,
                                                outline: 'none',
                                                boxSizing: 'border-box'
                                            }}
                                        />
                                        <div style={{fontFamily: EB.fMono, fontSize: 11, color: EB.sub, marginTop: 8}}>
                                            {tx("Đường dẫn hiện tại:", "Current path:")} <span style={{color: EB.ink, fontWeight: 600}}>{folderModal.targetFolder}</span>
                                        </div>
                                    </div>
                                )}

                                {(folderModal.mode === "MOVE_FOLDER" || folderModal.mode === "MOVE_QUIZZES") && (
                                    <div>
                                        <div style={{fontFamily: EB.fLabel, fontSize: 11, fontWeight: 700, letterSpacing: '0.08em', textTransform: 'uppercase', color: EB.sub, marginBottom: 10}}>
                                            {tx("Chọn thư mục đích", "Select destination folder")}
                                        </div>
                                        <div style={{
                                            maxHeight: 260,
                                            overflowY: 'auto',
                                            border: `1px solid ${EB.line}`,
                                            borderRadius: EB.radiusSm,
                                            background: EB.paper,
                                            padding: 6,
                                            display: 'flex',
                                            flexDirection: 'column',
                                            gap: 2
                                        }}>
                                            {allKnownFolders
                                                .filter((f: string) => {
                                                    const nf = normalizeFolderPath(f);
                                                    if (folderModal.mode === "MOVE_FOLDER" && folderModal.targetFolder) {
                                                        const targetNorm = normalizeFolderPath(folderModal.targetFolder);
                                                        if (nf === targetNorm || nf.startsWith(targetNorm + "/")) {
                                                            return false;
                                                        }
                                                    }
                                                    return true;
                                                })
                                                .map((rawF: string) => {
                                                    const f = normalizeFolderPath(rawF);
                                                    const depth = f === "Root" ? 0 : f.split("/").length - 1;
                                                    const isSelected = normalizeFolderPath(folderModal.selectedDest || "Root") === f;
                                                    const folderName = f === "Root" ? tx("Root (Gốc)", "Root") : f.split("/").pop();
                                                    return (
                                                        <button
                                                            key={f}
                                                            type="button"
                                                            onClick={() => setFolderModal({ ...folderModal, selectedDest: f })}
                                                            style={{
                                                                padding: '8px 12px',
                                                                paddingLeft: 12 + depth * 18,
                                                                borderRadius: EB.radiusSm,
                                                                border: `1px solid ${isSelected ? EB.accent : 'transparent'}`,
                                                                background: isSelected ? EB.accentWash : 'transparent',
                                                                color: isSelected ? EB.accent : EB.ink,
                                                                cursor: 'pointer',
                                                                display: 'flex',
                                                                alignItems: 'center',
                                                                justifyContent: 'space-between',
                                                                gap: 8,
                                                                fontSize: 13,
                                                                fontWeight: isSelected ? 700 : 500,
                                                                textAlign: 'left'
                                                            }}
                                                        >
                                                            <span style={{display: 'inline-flex', alignItems: 'center', gap: 7}}>
                                                                <Ico name="folder" size={14} color={isSelected ? EB.accent : EB.sub} />
                                                                <span>{folderName}</span>
                                                            </span>
                                                            {f !== "Root" && (
                                                                <span style={{fontFamily: EB.fMono, fontSize: 10, color: EB.sub, opacity: 0.7}}>
                                                                    {f}
                                                                </span>
                                                            )}
                                                        </button>
                                                    );
                                                })}
                                        </div>
                                        <div style={{fontFamily: EB.fMono, fontSize: 11, color: EB.sub, marginTop: 10}}>
                                            {tx("Đích đến:", "Destination:")} <span style={{color: EB.accent, fontWeight: 700}}>{folderModal.selectedDest || "Root"}</span>
                                        </div>
                                    </div>
                                )}
                            </div>

                            {/* Modal Footer */}
                            <div style={{
                                padding: '14px 20px',
                                borderTop: `1px solid ${EB.line}`,
                                display: 'flex',
                                justifyContent: 'flex-end',
                                gap: 10,
                                background: EB.wash
                            }}>
                                <button
                                    type="button"
                                    onClick={() => setFolderModal(null)}
                                    style={{
                                        padding: '8px 14px',
                                        borderRadius: EB.radiusSm,
                                        border: `1px solid ${EB.line}`,
                                        background: EB.sheet,
                                        color: EB.sub,
                                        fontSize: 12.5,
                                        fontWeight: 600,
                                        cursor: 'pointer'
                                    }}
                                >
                                    {tx("Hủy", "Cancel")}
                                </button>
                                <button
                                    type="button"
                                    onClick={() => {
                                        if (folderModal.mode === "CREATE") {
                                            if (folderModal.inputValue.trim()) {
                                                handleCreateFolder(folderModal.inputValue, folderModal.targetFolder || builderFolder);
                                            }
                                        } else if (folderModal.mode === "RENAME") {
                                            if (folderModal.inputValue.trim() && folderModal.targetFolder) {
                                                handleRenameFolder(folderModal.targetFolder, folderModal.inputValue);
                                            }
                                        } else if (folderModal.mode === "MOVE_FOLDER") {
                                            if (folderModal.targetFolder) {
                                                const dest = normalizeFolderPath(folderModal.selectedDest || "Root");
                                                const base = folderModal.targetFolder.split("/").pop() || "";
                                                const newPath = dest === "Root" ? `Root/${base}` : `${dest}/${base}`;
                                                handleMoveOrRenameFolder(folderModal.targetFolder, newPath);
                                            }
                                        } else if (folderModal.mode === "MOVE_QUIZZES") {
                                            handleMoveQuizzes(selectedQuizzes, normalizeFolderPath(folderModal.selectedDest || "Root"));
                                            setSelectedQuizzes([]);
                                        }
                                        setFolderModal(null);
                                    }}
                                    style={{
                                        padding: '8px 18px',
                                        borderRadius: EB.radiusSm,
                                        border: 'none',
                                        background: EB.accent,
                                        color: '#fff',
                                        fontSize: 12.5,
                                        fontWeight: 700,
                                        cursor: 'pointer'
                                    }}
                                >
                                    {folderModal.mode === "CREATE" && tx("Tạo thư mục", "Create folder")}
                                    {folderModal.mode === "RENAME" && tx("Lưu tên", "Save name")}
                                    {(folderModal.mode === "MOVE_FOLDER" || folderModal.mode === "MOVE_QUIZZES") && tx("Di chuyển", "Move")}
                                </button>
                            </div>
                        </div>
                    </div>
                )}
            </div>
            )
        })()}
