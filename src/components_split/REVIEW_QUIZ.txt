      const validateParaphrasePairs = (
          rawPairs: Array<{ questionTerm?: string; passageTerm?: string; note?: string }>,
          qText: string,
          ansText: string,
          sourceText: string
      ) => {
          if (!Array.isArray(rawPairs) || !rawPairs.length) return [];
          const normText = (s: any) => String(s || "").toLowerCase().replace(/['"“”‘’]/g, "").replace(/[.,!?;:()\[\]{}]/g, " ").replace(/\s+/g, " ").trim();
          const stripArticles = (s: string) => s.replace(/\b(the|a|an|of|to|in|for|on|with|at|by|from|and|or)\b\s+/g, " ").replace(/\s+/g, " ").trim();

          const stopWords = new Set(["the", "a", "an", "of", "to", "in", "for", "on", "with", "at", "by", "from", "and", "or"]);

          return rawPairs.map(p => {
              let qTerm = String(p.questionTerm || "").trim();
              let pTerm = String(p.passageTerm || "").trim();
              if (!qTerm || !pTerm) return null;

              let nQ = normText(qTerm);
              let nP = normText(pTerm);

              // Auto-swap if inverted
              if (!qHay.includes(nQ) && sHay.includes(nQ) && qHay.includes(nP) && !sHay.includes(nP)) {
                  const temp = qTerm;
                  qTerm = pTerm;
                  pTerm = temp;
                  nQ = normText(qTerm);
                  nP = normText(pTerm);
              }

              // Rule A: True Source Membership
              const qWords = stripArticles(nQ).split(" ").filter(w => w.length >= 3 && !stopWords.has(w));
              const pWords = stripArticles(nP).split(" ").filter(w => w.length >= 3 && !stopWords.has(w));
              const qInStem = qHay ? (qHay.includes(nQ) || (qWords.length >= 2 && qWords.every(w => qHay.includes(w)))) : true;
              const pInSource = sHay ? (sHay.includes(nP) || (pWords.length >= 2 && pWords.every(w => sHay.includes(w)))) : true;

              if (!qInStem || !pInSource) return null;

              // Rule B: No Identical or Trivial Substring Echoes
              const cleanQ = stripArticles(nQ);
              const cleanP = stripArticles(nP);
              if (cleanQ === cleanP) return null;

              const setQ = new Set(cleanQ.split(" ").filter(Boolean));
              const setP = new Set(cleanP.split(" ").filter(Boolean));
              if (!setQ.size || !setP.size) return null;

              const intersection = new Set([...setQ].filter(x => setP.has(x)));
              const maxWords = Math.max(setQ.size, setP.size);
              const minWords = Math.min(setQ.size, setP.size);
              if (maxWords > 0 && intersection.size / maxWords >= 0.7) return null;
              if (minWords > 0 && intersection.size / minWords >= 0.85 && (cleanQ.includes(cleanP) || cleanP.includes(cleanQ))) return null;

              return { questionTerm: qTerm, passageTerm: pTerm, note: p.note || "" };
          }).filter(Boolean) as Array<{ questionTerm: string; passageTerm: string; note: string }>;

          if (!match) return { start: -1 };

      const rvJumpToTranscript = (timestamp: string, quote = "") => {

                  for (let textNode = walker.nextNode() as Text | null; textNode; textNode = walker.nextNode() as Text | null) {
                      const length = textNode.data.length;
                      nodes.push({ node: textNode, start: cursor, end: cursor + length });
                      cursor += length;
                  }
                  const start = match.index;
                  const end = start + match[0].length;
                  nodes.slice().reverse().forEach(({ node, start: nodeStart, end: nodeEnd }) => {
                      const from = Math.max(start, nodeStart);
                      const to = Math.min(end, nodeEnd);
                      if (from >= to || !node.parentNode) return;
                      const fragment = document.createDocumentFragment();
                      const localStart = from - nodeStart;
                      const localEnd = to - nodeStart;
                      if (localStart) fragment.appendChild(document.createTextNode(node.data.slice(0, localStart)));

      const rvFormatOptionWithLetter = (q: any, rawValue: any): string => {

      const rvRenderInlineTokens = (textStr: string, keyPrefix: string, question?: any): React.ReactNode[] => {
          const tokenRegex = /(\*\*[^*\n]+\*\*|\*(?!\s)[^*\n]+(?<!\s)\*|_(?!\s)[^_\n]+(?<!\s)_|\[\d{1,2}:\d{2}(?::\d{2})?(?:\s*(?:[-–—~]|to)\s*\d{1,2}:\d{2}(?::\d{2})?)?\]|"[^"\n]{2,260}"|“[^”\n]{2,260}”)/g;
          const nodes: React.ReactNode[] = [];
          let lastIdx = 0;
          let match: RegExpExecArray | null;

          while ((match = tokenRegex.exec(textStr)) !== null) {
              if (match.index > lastIdx) {
                  nodes.push(textStr.slice(lastIdx, match.index));
              }
              const tok = match[0];
              const k = `${keyPrefix}-${match.index}`;
              if (tok.startsWith("**") && tok.endsWith("**")) {
                  const inner = tok.slice(2, -2);
                  nodes.push(<strong key={k} className="rv-md-bold">{rvRenderInlineTokens(inner, `${k}-b`, question)}</strong>);
              } else if ((tok.startsWith("*") && tok.endsWith("*")) || (tok.startsWith("_") && tok.endsWith("_"))) {
                  const inner = tok.slice(1, -1);
                  nodes.push(<em key={k} className="rv-md-italic">{inner}</em>);
              } else if (tok.startsWith("[") && tok.endsWith("]")) {
                  const tsLabel = tok.slice(1, -1).trim();
                  nodes.push(
                      <button
                          key={k}
                          type="button"
                          className="micro-btn-audio"
                          style={{ margin: "0 3px", verticalAlign: "baseline", height: 22, padding: "0 6px", fontSize: 11 }}
                          onClick={() => {
                              const { start, end } = rvTimestampRangeSeconds(tsLabel);
                              if (question) rvSeekManualTimestamp({ startTime: start, endTime: end, label: tsLabel }, question);
                              else rvSeek(tsLabel);
                          }}
                          title={`Phát audio tại ${tsLabel}`}
                      >
                          <svg width="10" height="10" viewBox="0 0 24 24" fill="currentColor"><polygon points="6 4 20 12 6 20 6 4" /></svg>
                          <span>{tsLabel}</span>
                      </button>
                  );
              } else if ((tok.startsWith('"') && tok.endsWith('"')) || (tok.startsWith("“") && tok.endsWith("”"))) {
                  const quoteInner = tok.slice(1, -1).replace(/^"+|"+$/g, "").trim();
                  const canJump = quoteInner.length >= 12 && quoteInner.split(/\s+/).length >= 3;
                  if (canJump) {
                      nodes.push(
                          <em
                              key={k}
                              className="rv-md-quote is-clickable"
                              onClick={() => {
                                  if (rvActiveIsListening) rvJumpToTranscript("", quoteInner);
                                  else rvJumpToEvidence(quoteInner);
                              }}
                              title="Nhấn để định vị câu trích dẫn trong bài"
                          >
                              “{quoteInner}”
                          </em>
                      );
                  } else {
                      nodes.push(<em key={k} className="rv-md-quote">“{quoteInner}”</em>);
                  }
              } else {
                  nodes.push(tok);
              }
              lastIdx = tokenRegex.lastIndex;
          }
          if (lastIdx < textStr.length) {
              nodes.push(textStr.slice(lastIdx));
          }
          return nodes;
      };

                  const segments = text.split(/(\[\d+\](?:\s*(?:_{2,}|\.{3,}|…+))?)/g);

                      const m = seg.match(/^\[(\d+)\]/);
                      if (m) {
                          const q = byNumber.get(Number(m[1]));
                          if (q) {
                              if (hostQuestions) {
                                  if (!hostQuestions.some(item => item.id === q.id)) hostQuestions.push(q);

                              return <React.Fragment key={`${keyPrefix}-slot-${sIdx}`}>{rvRenderInlineSlotBox(q, isDashedDrag, true)}</React.Fragment>;
                          }
                      }
                      return seg;

                                                                  ? (retryChoice !== undefined && retryChoice !== "" && Number(retryChoice) === oIdx)
                                                                  : (Array.isArray(retryChoice) && retryChoice.map(Number).includes(oIdx));

                                                              return (
                                                                  <li
                                                                      key={oIdx}
                                                                      className={`rv-mcq-opt ${!rvRetryMode && isCorrectOpt ? "is-correct" : !rvRetryMode && isChosenOpt ? "is-wrong" : ""}`}
                                                                      onClick={() => {
                                                                          if (!rvRetryMode || qStatus === "correct") return;
                                                                          if (q.type === "CHOICE") {
                                                                              setRvRetryAnswers(prev => ({ ...prev, [q.id]: oIdx }));
                                                                          } else {
                                                                              const curArr = Array.isArray(rvRetryAnswers[q.id]) ? rvRetryAnswers[q.id].map(Number) : [];
                                                                              const nextArr = curArr.includes(oIdx) ? curArr.filter((x: number) => x !== oIdx) : [...curArr, oIdx];
                                                                              setRvRetryAnswers(prev => ({ ...prev, [q.id]: nextArr }));
                                                                          }
                                                                      }}
                                                                      style={{
                                                                          cursor: rvRetryMode && qStatus !== "correct" ? "pointer" : "default",
                                                                          ...(rvRetryMode && isRetrySelected ? { borderColor: "#D97706", background: "#FEF3C7", fontWeight: 700 } : {}),
                                                                      }}
                                                                  >
                                                                      <div>
                                                                          <strong className="rv-mono" style={{ marginRight: 8 }}>{optLetter}.</strong>
                                                                          <span style={{ textDecoration: !rvRetryMode && isChosenOpt && !isCorrectOpt ? "line-through" : "none" }} dangerouslySetInnerHTML={{ __html: formatContent(cleanText) }} />
                                                                      </div>
                                                                      {!rvRetryMode && isCorrectOpt && <span className="rv-mcq-tag">(Đáp án đúng)</span>}
                                                                      {!rvRetryMode && isChosenOpt && !isCorrectOpt && <span className="rv-mcq-tag">(Bạn chọn - Sai)</span>}
                                                                      {rvRetryMode && isRetrySelected && <span className="rv-mcq-tag" style={{ color: "#92400E" }}>(Đang chọn lại)</span>}
                                                                  </li>
                                                              );
                                                          })}
                                                      </ul>
                                                  )}

                                                  {rvRenderInlineDrawer(qWithOpts, studentAns, multiOutcome ? { ...qWithOpts, correctAnswer: multiOutcome.correct } : qWithOpts)}
                                              </div>
                                          </React.Fragment>
                                      );
                                  });
                              })()}
                          </div>
                      </div>

                      {/* BOTTOM QUESTION PALETTE (Exam Room Mechanism) */}
                      <footer className="rv-footer-palette" style={{ height: 48, gap: 20 }}>
                          {rvSections.map((sec: any, sIdx: number) => {
                              const secQs = rvHasSections ? (sec?.questions || rvAllQuestions.filter((q: any) => rvSectionOf(q) === sIdx)) : rvAllQuestions;
                              const secCorrect = secQs.filter((q: any) => rvQuestionStatus(q) === "correct").length;
                              const isActive = rvActiveIdx === sIdx;
                              const label = rvUsePartLabels ? `Part ${sIdx + 1}` : `Passage ${sIdx + 1}`;
                              const allCorrect = secQs.length > 0 && secCorrect === secQs.length;
                              const pct = secQs.length ? Math.round((secCorrect / secQs.length) * 100) : 0;
                              const switchSection = () => {
                                  setReviewSectionIdx(sIdx);
                                  setReviewActiveQuestionId(null);
                                  const rightPane = document.querySelector(".rv-pane-right");
                                  if (rightPane) rightPane.scrollTop = 0;
                                  const leftPane = document.querySelector(".rv-pane-left-scroll");
                                  if (leftPane) leftPane.scrollTop = 0;
                              };
                              return (
                                  <div
                                      key={sIdx}
                                      onClick={!isActive ? switchSection : undefined}
                                      style={{
                                          display: "inline-flex",
                                          alignItems: "center",
                                          gap: 10,
                                          flexShrink: 0,
                                          cursor: !isActive ? "pointer" : "default",
                                          padding: "4px 6px",
                                          borderRadius: 6,
                                      }}
                                  >
                                      <div style={{ display: "flex", flexDirection: "column", alignItems: "stretch", gap: 4, flexShrink: 0 }}>
                                          <div style={{ height: 3, width: "100%", minWidth: 42, background: "#E6DFD3", borderRadius: 2, overflow: "hidden" }}>
                                              <div style={{ height: "100%", width: `${pct}%`, background: allCorrect ? "#16A34A" : "#1C1917", transition: "width .25s ease" }} />
                                          </div>
                                          <button
                                              type="button"
                                              onClick={switchSection}
                                              style={{
                                                  border: "none",
                                                  background: "transparent",
                                                  padding: 0,
                                                  fontSize: 13,
                                                  fontWeight: isActive ? 800 : 600,
                                                  color: isActive ? "#1C1917" : "#57534E",
                                                  cursor: "pointer",
                                                  display: "inline-flex",
                                                  alignItems: "center",
                                                  justifyContent: "center",
                                                  gap: 5,
                                                  whiteSpace: "nowrap",
                                              }}
                                          >
                                              {allCorrect && (
                                                  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="#16A34A" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round" style={{ flexShrink: 0 }}>
                                                      <polyline points="20 6 9 17 4 12" />
                                                  </svg>
                                              )}
                                              <span>{label}</span>
                                          </button>
                                      </div>
                                      {isActive ? (
                                          <div style={{ display: "inline-flex", alignItems: "center", gap: 5 }}>
                                              {secQs.map((q: any) => {
                                                  const qNum = rvQuestionNumber(q);
                                                  const st = rvQuestionStatus(q);
                                                  const isCur = reviewActiveQuestionId === q.id;
                                                  return (
                                                      <button
                                                          key={q.id}
                                                          type="button"
                                                          className={`rv-palette-btn ${st}`}
                                                          style={isCur ? { outline: "2px solid #1C1917", outlineOffset: 1 } : undefined}
                                                          onClick={(e) => {
                                                              e.stopPropagation();
                                                              rvJumpToQuestionFromMatrix(q, qNum);
                                                          }}
                                                          title={`Câu ${qNum}`}
                                                      >
                                                          {qNum}
                                                      </button>
                                                  );
                                              })}
                                          </div>
                                      ) : (
                                          <span style={{ fontSize: 12, color: "#78716C", whiteSpace: "nowrap" }}>
                                              {secCorrect} of {secQs.length}
                                          </span>
                                      )}
                                  </div>
                              );
                          })}
                      </footer>
                  </>
              )}

              {/* MINIMALIST FREE-FORM "TRỢ LÝ AI" SLIDE-OVER DRAWER */}
              {copilotOpen && (
                  <aside
                      style={{
                          position: "fixed",
                          top: 0,
                          right: 0,
                          width: 380,
                          maxWidth: "100vw",
                          height: "100vh",
                          background: "#FFFFFF",
                          borderLeft: "1px solid #E6DFD3",
                          boxShadow: "-8px 0 24px rgba(28, 25, 23, 0.08)",
                          zIndex: 200,
                          display: "flex",
                          flexDirection: "column",
                      }}
                  >
                      {/* Drawer Header */}
                      <div style={{ height: 52, padding: "0 16px", borderBottom: "1px solid #E6DFD3", display: "flex", alignItems: "center", justifyContent: "space-between" }}>
                          <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
                              <span style={{ width: 8, height: 8, borderRadius: "50%", background: "#16A34A", display: "inline-block" }} />
                              <strong style={{ fontSize: 14, fontWeight: 800, color: "#1C1917" }}>Trợ lý AI</strong>
                          </div>
                          <button
                              type="button"
                              onClick={() => setCopilotOpen(false)}
                              style={{ border: "none", background: "transparent", color: "#57534E", cursor: "pointer", padding: 4, display: "inline-flex" }}
                              aria-label="Đóng Trợ lý AI"
                          >
                              <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.2"><line x1="18" y1="6" x2="6" y2="18" /><line x1="6" y1="6" x2="18" y2="18" /></svg>
                          </button>
                      </div>

                      {/* Drawer Chat Thread */}
                      <div
                          ref={copilotChatBoxRef}
                          style={{
                              flex: 1,
                              overflowY: "auto",
                              padding: 16,
                              display: "flex",
                              flexDirection: "column",
                              gap: 10,
                              background: "#FAF7F2",
                          }}
                      >
                          {(copilotMessages.length > 0 ? copilotMessages : [{ role: "copilot" as const, text: "Chào bạn, bạn cần hỗ trợ gì?" }]).map((msg, mIdx) => {
                              const isUser = msg.role === "user";
                              return (
                                  <div
                                      key={mIdx}
                                      style={{
                                          alignSelf: isUser ? "flex-end" : "flex-start",
                                          maxWidth: "90%",
                                          background: isUser ? "#1C1917" : "#F4EFE6",
                                          color: isUser ? "#FFFFFF" : "#1C1917",
                                          border: isUser ? "none" : "1px solid #E6DFD3",
                                          borderRadius: isUser ? "12px 12px 2px 12px" : "12px 12px 12px 2px",
                                          padding: "10px 13px",
                                          fontSize: 13.5,
                                          lineHeight: 1.55,
                                          wordBreak: "break-word",
                                      }}
                                  >
                                      {isUser ? msg.text : rvRenderRichText(msg.text)}
                                  </div>
                              );
                          })}
                          {copilotLoading && (
                              <div style={{ alignSelf: "flex-start", background: "#F4EFE6", border: "1px solid #E6DFD3", borderRadius: "12px 12px 12px 2px", padding: "8px 12px", fontSize: 12.5, color: "#57534E" }}>
                                  Đang trả lời...
                              </div>
                          )}
                      </div>

                      {/* Drawer Input Footer */}
                      <div style={{ padding: 12, borderTop: "1px solid #E6DFD3", background: "#FFFFFF", display: "flex", gap: 8 }}>
                          <input
                              type="text"
                              value={copilotInput}
                              onChange={e => setCopilotInput(e.target.value)}
                              onKeyDown={e => { if (e.key === "Enter") void handleSendCopilotMessage(); }}
                              placeholder="Hỏi bất kỳ câu nào (VD: Giải thích câu 5, tại sao câu 12 không chọn B)..."
                              style={{
                                  flex: 1,
                                  height: 38,
                                  padding: "0 12px",
                                  borderRadius: 8,
                                  border: "1px solid #D4CBB8",
                                  background: "#FAF7F2",
                                  color: "#1C1917",
                                  fontSize: 13,
                                  outline: "none",
                              }}
                          />
                          <button
                              type="button"
                              disabled={copilotLoading || !copilotInput.trim()}
                              onClick={() => void handleSendCopilotMessage()}
                              style={{
                                  height: 38,
                                  padding: "0 14px",
                                  borderRadius: 8,
                                  border: "none",
                                  background: "#1C1917",
                                  color: "#FFFFFF",
                                  fontSize: 13,
                                  fontWeight: 700,
                                  cursor: copilotLoading || !copilotInput.trim() ? "default" : "pointer",
                                  opacity: copilotLoading || !copilotInput.trim() ? 0.5 : 1,
                              }}
                          >
                              Gửi
                          </button>
                      </div>
                  </aside>
              )}
                  {/* Floating Auto-Flip Explanation Popover for Matching Headings in Left Pane */}
                  {activeHeadingPopover && (() => {
                      const hq = rvAllQuestions.find((q: any) => q.id === activeHeadingPopover.qId);
                      if (!hq) return null;
                      const qNum = rvQuestionNumber(hq);
                      const manual = rvManualExplanationOf(hq);
                      const aiState = explainMap[hq.id];
                      const studentAns = rvSubmittedAnswers?.[hq.id];
                      const rect = activeHeadingPopover.anchorRect;
                      const isTeacher = activeHeadingPopover.mode === "teacher";

                      return (
                          <div
                              id="rv-heading-popover"
                              style={{
                                  position: "fixed",
                                  top: rect?.top ? `${rect.top}px` : "auto",
                                  bottom: rect?.bottom ? `${rect.bottom}px` : "auto",
                                  left: `${rect?.left || 20}px`,
                                  width: `${rect?.width || 480}px`,
                                  maxHeight: `${rect?.height || 360}px`,
                                  zIndex: 9999,
                                  background: "#FFFFFF",
                                  borderRadius: 10,
                                  boxShadow: "0 10px 25px -5px rgba(0, 0, 0, 0.2), 0 8px 10px -6px rgba(0, 0, 0, 0.1), 0 0 0 1px rgba(0, 0, 0, 0.08)",
                                  display: "flex",
                                  flexDirection: "column",
                                  overflow: "hidden",
                                  animation: "rvFadeIn 0.15s ease-out",
                              }}
                              onClick={(e) => e.stopPropagation()}
                          >
                              <div
                                  style={{
                                      padding: "10px 14px",
                                      background: isTeacher ? "#FFFBEB" : "#F8FAFC",
                                      borderBottom: "1px solid #E2E8F0",
                                      display: "flex",
                                      alignItems: "center",
                                      justifyContent: "space-between",
                                      flexShrink: 0,
                                  }}
                              >
                                  <div style={{ display: "flex", alignItems: "center", gap: 7, fontWeight: 800, fontSize: 13, color: isTeacher ? "#92400E" : "#1E293B" }}>
                                      <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke={isTeacher ? "#D97706" : "#2563EB"} strokeWidth="2.2" strokeLinecap="round" strokeLinejoin="round">
                                          <path d="M9 18h6" />
                                          <path d="M10 22h4" />
                                          <path d="M15.09 14c.18-.98.65-1.74 1.41-2.5A4.65 4.65 0 0 0 18 8 6 6 0 0 0 6 8c0 1 .23 2.23 1.5 3.5A4.61 4.61 0 0 1 8.91 14" />
                                      </svg>
                                      <span>{isTeacher ? `GIẢI THÍCH GIÁO VIÊN — CÂU ${qNum}` : `GIẢI THÍCH CHI TIẾT — CÂU ${qNum}`}</span>
                                  </div>
                                  <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
                                      {!isTeacher && !aiState?.loading && (
                                          <button
                                              type="button"
                                              className="rv-drawer-reanalyze-btn"
                                              onClick={() => void handleAiExplain(hq, studentAns, reviewQuiz.quiz)}
                                              style={{ fontSize: 11.5, padding: "3px 8px" }}
                                          >
                                              <svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.4" strokeLinecap="round" strokeLinejoin="round" style={{ marginRight: 4 }}>
                                                  <path d="M3 12a9 9 0 1 0 9-9 9.75 9.75 0 0 0-6.74 2.74L3 8" />
                                                  <path d="M3 3v5h5" />
                                              </svg>
                                              Phân tích lại
                                          </button>
                                      )}
                                      <button
                                          type="button"
                                          className="rv-drawer-close"
                                          onClick={() => setActiveHeadingPopover(null)}
                                          aria-label="Đóng"
                                          style={{ width: 22, height: 22, display: "flex", alignItems: "center", justifyContent: "center", border: "none", background: "transparent", cursor: "pointer", borderRadius: 4, color: "#64748B" }}
                                      >
                                          <svg width="13" height="13" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.2"><line x1="18" y1="6" x2="6" y2="18" /><line x1="6" y1="6" x2="18" y2="18" /></svg>
                                      </button>
                                  </div>
                              </div>

                              <div style={{ padding: "12px 14px", overflowY: "auto", flex: 1, fontSize: 13, lineHeight: 1.6 }}>
                                  {isTeacher ? (
                                      manual?.rawText ? (
                                          rvRenderRichText(manual.rawText, hq)
                                      ) : (
                                          <div style={{ color: "#78716C", fontStyle: "italic", textAlign: "center", padding: "16px 0" }}>
                                              Chưa có ghi chú thủ công của Giáo viên cho câu này.
                                          </div>
                                      )
                                  ) : (
                                      (aiState?.structured || aiState?.text) ? (
                                          rvRenderDetailedAiExplanation(aiState, hq, studentAns)
                                      ) : aiState?.loading ? (
                                          <div role="status" aria-live="polite" style={{ padding: "6px 0 2px" }}>
                                              <style>{"@keyframes rvSpin{to{transform:rotate(360deg)}}@keyframes rvPulse{0%,100%{opacity:.45}50%{opacity:1}}@keyframes rvFadeIn{from{opacity:0;transform:translateY(-4px)}to{opacity:1;transform:none}}"}</style>
                                              <div style={{ display: "flex", alignItems: "center", gap: 8, color: "#475569", fontSize: 12.5, fontWeight: 600, marginBottom: 12 }}>
                                                  <span style={{ width: 14, height: 14, borderRadius: "50%", border: "2px solid #CBD5E1", borderTopColor: "#2563EB", animation: "rvSpin 0.8s linear infinite", display: "inline-block" }} />
                                                  Đang đọc đoạn văn và đối chiếu heading...
                                              </div>
                                              {[92, 78, 64].map((w, k) => (
                                                  <div key={k} style={{ height: 10, width: `${w}%`, borderRadius: 4, background: "#E2E8F0", marginBottom: 8, animation: `rvPulse 1.2s ease-in-out ${k * 0.15}s infinite` }} />
                                              ))}
                                          </div>
                                      ) : (
                                          <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", gap: 10, padding: "6px 0", color: "#64748B", fontSize: 12.5 }}>
                                              <span>Chưa tải được phân tích cho đoạn này.</span>
                                              <button
                                                  type="button"
                                                  className="rv-drawer-switch-ai"
                                                  onClick={() => void handleAiExplain(hq, studentAns, reviewQuiz.quiz)}
                                              >
                                                  Thử lại
                                              </button>
                                          </div>
                                      )
                                  )}
                              </div>
                          </div>
                      );
                  })()}
          </div>
      );
  }
